Примеры кода¶

В составе исходного кода Reticulum поставляется несколько примеров. Вы можете использовать их, чтобы научиться писать собственные программы.

Минимальный (Minimal)¶

Пример Minimal показывает самый минимальный объём кода, необходимый для подключения к сети Reticulum из вашей программы. Буквально за пять строк кода вы запустите Сетевой Стек Reticulum, готовый пропускать трафик в вашей программе.

##########################################################
# Этот пример RNS показывает минимальную настройку,     #
# которая запустит Сетевой Стек Reticulum, сгенерирует  #
# новое назначение и позволит пользователю отправить    #
# объявление.                                           #
##########################################################

import argparse
import sys
import RNS

# Давайте определим имя приложения. Мы будем использовать
# его для всех создаваемых нами назначений. Поскольку
# этот базовый пример является частью набора примеров
# утилит, мы поместим их все в пространство имён
# приложения "example_utilities"
APP_NAME = "example_utilities"

# Эта инициализация выполняется при запуске программы
def program_setup(configpath):
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)
    
    # Создаём случайную новую идентичность для нашего примера
    identity = RNS.Identity()

    # Используя только что созданную идентичность, мы создаём
    # назначение. Назначения - это конечные точки в Reticulum,
    # которые можно адресовать и с которыми можно общаться.
    # Назначения также могут объявлять о своём существовании,
    # что позволит сети узнать, что они достижимы, и
    # автоматически создавать пути к ним из любой другой
    # точки сети.
    destination = RNS.Destination(
        identity,
        RNS.Destination.IN,
        RNS.Destination.SINGLE,
        APP_NAME,
        "minimalsample"
    )

    # Мы настраиваем назначение так, чтобы оно автоматически
    # доказывало получение всех пакетов, адресованных ему.
    # Благодаря этому RNS будет автоматически генерировать
    # доказательство для каждого входящего пакета и
    # передавать его обратно отправителю. Это позволит
    # любому, кто пытается связаться с назначением, узнать,
    # было ли его сообщение получено корректно.
    destination.set_proof_strategy(RNS.Destination.PROVE_ALL)
    
    # Всё готово!
    # Давайте передадим управление циклу объявлений
    announceLoop(destination)


def announceLoop(destination):
    # Сообщаем пользователю, что всё готово
    RNS.log(
        "Minimal example "+
        RNS.prettyhexrep(destination.hash)+
        " running, hit enter to manually send an announce (Ctrl-C to quit)"
    )

    # Входим в цикл, который работает до тех пор, пока
    # пользователь не выйдет. Если пользователь нажимает
    # enter, мы объявим наше серверное назначение в сети,
    # что позволит клиентам узнать, как создавать
    # сообщения, направленные к нему.
    while True:
        entered = input()
        destination.announce()
        RNS.log("Sent announce from "+RNS.prettyhexrep(destination.hash))


##########################################################
#### Запуск Программы ###################################
##########################################################

# Эта часть программы выполняется при запуске,
# разбирает ввод от пользователя, а затем запускает
# нужный режим работы.
if __name__ == "__main__":
    try:
        parser = argparse.ArgumentParser(
            description="Minimal example to start Reticulum and create a destination"
        )

        parser.add_argument(
            "--config",
            action="store",
            default=None,
            help="path to alternative Reticulum config directory",
            type=str
        )

        args = parser.parse_args()

        if args.config:
            configarg = args.config
        else:
            configarg = None

        program_setup(configarg)

    except KeyboardInterrupt:
        print("")
        sys.exit(0)

Объявление (Announce)¶

Пример Announce основывается на предыдущем примере и показывает, как объявить назначение в сети и как позволить вашей программе получать уведомления об объявлениях от релевантных назначений.

##########################################################
# Этот пример RNS демонстрирует настройку callback-     #
# функций для объявлений, которые позволят приложению   #
# получать уведомления при поступлении релевантного     #
# объявления.                                           #
##########################################################

import argparse
import random
import sys
import RNS

# Давайте определим имя приложения. Мы будем использовать
# его для всех создаваемых нами назначений. Поскольку
# этот базовый пример является частью набора примеров
# утилит, мы поместим их все в пространство имён
# приложения "example_utilities"
APP_NAME = "example_utilities"

# Мы инициализируем два списка строк для использования
# в качестве app_data
fruits = ["Peach", "Quince", "Date", "Tangerine", "Pomelo", "Carambola", "Grape"]
noble_gases = ["Helium", "Neon", "Argon", "Krypton", "Xenon", "Radon", "Oganesson"]

# Эта инициализация выполняется при запуске программы
def program_setup(configpath):
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)
    
    # Создаём случайную новую идентичность для нашего примера
    identity = RNS.Identity()

    # Используя только что созданную идентичность, мы создаём
    # два назначения в пространстве приложения
    # "example_utilities.announcesample".
    #
    # Назначения - это конечные точки в Reticulum, которые
    # можно адресовать и с которыми можно общаться.
    # Назначения также могут объявлять о своём существовании,
    # что позволит сети узнать, что они достижимы, и
    # автоматически создавать пути к ним из любой другой
    # точки сети.
    destination_1 = RNS.Destination(
        identity,
        RNS.Destination.IN,
        RNS.Destination.SINGLE,
        APP_NAME,
        "announcesample",
        "fruits"
    )

    destination_2 = RNS.Destination(
        identity,
        RNS.Destination.IN,
        RNS.Destination.SINGLE,
        APP_NAME,
        "announcesample",
        "noble_gases"
    )

    # Мы настраиваем назначения так, чтобы они автоматически
    # доказывали получение всех пакетов. Благодаря этому
    # RNS будет автоматически генерировать доказательство
    # для каждого входящего пакета и передавать его обратно
    # отправителю. Это позволит любому, кто пытается
    # связаться с назначением, узнать, было ли его сообщение
    # получено корректно.
    destination_1.set_proof_strategy(RNS.Destination.PROVE_ALL)
    destination_2.set_proof_strategy(RNS.Destination.PROVE_ALL)

    # Мы создаём обработчик объявлений и настраиваем его
    # так, чтобы он запрашивал только объявления от
    # "example_utilities.announcesample.fruits".
    # Попробуйте изменить фильтр и посмотрите, что произойдёт.
    announce_handler = ExampleAnnounceHandler(
        aspect_filter="example_utilities.announcesample.fruits"
    )

    # Регистрируем обработчик объявлений в Reticulum
    RNS.Transport.register_announce_handler(announce_handler)
    
    # Всё готово!
    # Давайте передадим управление циклу объявлений
    announceLoop(destination_1, destination_2)


def announceLoop(destination_1, destination_2):
    # Сообщаем пользователю, что всё готово
    RNS.log("Announce example running, hit enter to manually send an announce (Ctrl-C to quit)")

    # Входим в цикл, который работает до тех пор, пока
    # пользователь не выйдет. Если пользователь нажимает
    # enter, мы объявим наше серверное назначение в сети,
    # что позволит клиентам узнать, как создавать
    # сообщения, направленные к нему.
    while True:
        entered = input()
        
        # Случайным образом выбираем фрукт
        fruit = fruits[random.randint(0,len(fruits)-1)]

        # Отправляем объявление, включая app_data
        destination_1.announce(app_data=fruit.encode("utf-8"))
        RNS.log(
            "Sent announce from "+
            RNS.prettyhexrep(destination_1.hash)+
            " ("+destination_1.name+")"
        )

        # Случайным образом выбираем благородный газ
        noble_gas = noble_gases[random.randint(0,len(noble_gases)-1)]

        # Отправляем объявление, включая app_data
        destination_2.announce(app_data=noble_gas.encode("utf-8"))
        RNS.log(
            "Sent announce from "+
            RNS.prettyhexrep(destination_2.hash)+
            " ("+destination_2.name+")"
        )

# Нам нужно определить класс обработчика объявлений,
# которому Reticulum сможет отправлять сообщения при
# поступлении объявления.
class ExampleAnnounceHandler:
    # Метод инициализации принимает необязательный аргумент
    # aspect_filter. Если aspect_filter установлен в None,
    # все объявления будут передаваться экземпляру. Если
    # нужны только некоторые объявления, его можно
    # установить в строку аспекта.
    def __init__(self, aspect_filter=None):
        self.aspect_filter = aspect_filter

    # Этот метод будет вызываться Транспортной системой
    # Reticulum при поступлении объявления, которое
    # соответствует настроенному фильтру аспектов.
    # Фильтры должны быть конкретными и не могут
    # использовать подстановочные знаки.
    def received_announce(self, destination_hash, announced_identity, app_data):
        RNS.log(
            "Received an announce from "+
            RNS.prettyhexrep(destination_hash)
        )

        if app_data:
            RNS.log(
                "The announce contained the following app data: "+
                app_data.decode("utf-8")
            )

##########################################################
#### Запуск Программы ###################################
##########################################################

# Эта часть программы выполняется при запуске,
# разбирает ввод от пользователя, а затем запускает
# нужный режим работы.
if __name__ == "__main__":
    try:
        parser = argparse.ArgumentParser(
            description="Reticulum example that demonstrates announces and announce handlers"
        )

        parser.add_argument(
            "--config",
            action="store",
            default=None,
            help="path to alternative Reticulum config directory",
            type=str
        )

        args = parser.parse_args()

        if args.config:
            configarg = args.config
        else:
            configarg = None

        program_setup(configarg)

    except KeyboardInterrupt:
        print("")
        sys.exit(0)

Широковещательная передача (Broadcast)¶

Пример Broadcast показывает, как передавать открытые широковещательные сообщения по сети.

##########################################################
# Этот пример RNS демонстрирует передачу незашифрованной#
# информации на любые прослушивающие назначения.         #
##########################################################

import sys
import argparse
import RNS

# Давайте определим имя приложения. Мы будем использовать
# его для всех создаваемых нами назначений. Поскольку
# этот базовый пример является частью набора примеров
# утилит, мы поместим их все в пространство имён
# приложения "example_utilities"
APP_NAME = "example_utilities"

# Эта инициализация выполняется при запуске программы
def program_setup(configpath, channel=None):
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)
    
    # Если пользователь не выбрал "канал", мы используем
    # канал по умолчанию с именем "public_information".
    # Этот "канал" добавляется в пространство имён
    # назначения, поэтому пользователь может выбирать
    # различные каналы для широковещания.
    if channel == None:
        channel = "public_information"

    # Мы создаём PLAIN (открытое) назначение. Это
    # незашифрованная конечная точка, которую любой может
    # слушать и на которую любой может отправлять информацию.
    broadcast_destination = RNS.Destination(
        None,
        RNS.Destination.IN,
        RNS.Destination.PLAIN,
        APP_NAME,
        "broadcast",
        channel
    )

    # Мы указываем callback-функцию, которая будет
    # вызываться каждый раз, когда назначение получает
    # данные.
    broadcast_destination.set_packet_callback(packet_callback)
    
    # Всё готово!
    # Передаём управление основному циклу
    broadcastLoop(broadcast_destination)

def packet_callback(data, packet):
    # Просто выводим полученные данные
    print("")
    print("Received data: "+data.decode("utf-8")+"\r\n> ", end="")
    sys.stdout.flush()

def broadcastLoop(destination):
    # Сообщаем пользователю, что всё готово
    RNS.log(
        "Broadcast example "+
        RNS.prettyhexrep(destination.hash)+
        " running, enter text and hit enter to broadcast (Ctrl-C to quit)"
    )

    # Входим в цикл, который работает до тех пор, пока
    # пользователь не выйдет. Если пользователь нажимает
    # enter, мы отправим информацию, которую он ввёл.
    while True:
        print("> ", end="")
        entered = input()

        if entered != "":
            data    = entered.encode("utf-8")
            packet  = RNS.Packet(destination, data)
            packet.send()



##########################################################
#### Запуск Программы ###################################
##########################################################

# Эта часть программы выполняется при запуске,
# разбирает ввод от пользователя, а затем запускает
# программу.
if __name__ == "__main__":
    try:
        parser = argparse.ArgumentParser(
            description="Reticulum example demonstrating sending and receiving broadcasts"
        )

        parser.add_argument(
            "--config",
            action="store",
            default=None,
            help="path to alternative Reticulum config directory",
            type=str
        )

        parser.add_argument(
            "--channel",
            action="store",
            default=None,
            help="broadcast channel name",
            type=str
        )

        args = parser.parse_args()

        if args.config:
            configarg = args.config
        else:
            configarg = None

        if args.channel:
            channelarg = args.channel
        else:
            channelarg = None

        program_setup(configarg, channelarg)

    except KeyboardInterrupt:
        print("")
        sys.exit(0)

Эхо (Echo)¶

Пример Echo демонстрирует обмен данными между двумя назначениями с использованием интерфейса Packet.

##########################################################
# Этот пример RNS демонстрирует простейшую клиент-      #
# серверную эхо-утилиту. Клиент может отправить эхо-    #
# запрос серверу, а сервер ответит, доказав получение   #
# пакета.                                               #
##########################################################

import argparse
import sys
import RNS

# Давайте определим имя приложения. Мы будем использовать
# его для всех создаваемых нами назначений. Поскольку
# этот пример эхо является частью набора примеров утилит,
# мы поместим их все в пространство имён приложения
# "example_utilities"
APP_NAME = "example_utilities"


##########################################################
#### Серверная Часть ####################################
##########################################################

# Эта инициализация выполняется, когда пользователь
# выбирает запуск в режиме сервера
def server(configpath):
    global reticulum

    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)
    
    # Создаём случайную новую идентичность для нашего
    # эхо-сервера
    server_identity = RNS.Identity()

    # Мы создаём назначение, которое клиенты могут
    # запрашивать. Мы хотим иметь возможность проверять
    # эхо-ответы для наших клиентов, поэтому создаём
    # "single" назначение, способное принимать зашифрованные
    # сообщения. Так клиент может отправить запрос и быть
    # уверенным, что никто, кроме этого назначения, не
    # сможет его прочитать. 
    echo_destination = RNS.Destination(
        server_identity,
        RNS.Destination.IN,
        RNS.Destination.SINGLE,
        APP_NAME,
        "echo",
        "request"
    )

    # Мы настраиваем назначение так, чтобы оно автоматически
    # доказывало получение всех пакетов, адресованных ему.
    # Благодаря этому RNS будет автоматически генерировать
    # доказательство для каждого входящего пакета и
    # передавать его обратно отправителю.
    echo_destination.set_proof_strategy(RNS.Destination.PROVE_ALL)
    
    # Указываем назначению, какую функцию в нашей программе
    # запускать при получении пакета. Мы делаем это, чтобы
    # выводить сообщение в лог при получении запроса сервером
    echo_destination.set_packet_callback(server_callback)

    # Всё готово!
    # Ждём запросов клиентов или ввода пользователя
    announceLoop(echo_destination)


def announceLoop(destination):
    # Сообщаем пользователю, что всё готово
    RNS.log(
        "Echo server "+
        RNS.prettyhexrep(destination.hash)+
        " running, hit enter to manually send an announce (Ctrl-C to quit)"
    )

    # Входим в цикл, который работает до тех пор, пока
    # пользователь не выйдет. Если пользователь нажимает
    # enter, мы объявим наше серверное назначение в сети,
    # что позволит клиентам узнать, как создавать сообщения,
    # направленные к нему.
    while True:
        entered = input()
        destination.announce()
        RNS.log("Sent announce from "+RNS.prettyhexrep(destination.hash))


def server_callback(message, packet):
    global reticulum
    
    # Сообщаем пользователю, что мы получили эхо-запрос и
    # собираемся отправить ответ запрашивающему. Отправка
    # доказательства обрабатывается автоматически, так как
    # мы настроили назначение доказывать все входящие
    # пакеты.

    reception_stats = ""
    if reticulum.is_connected_to_shared_instance:
        reception_rssi = reticulum.get_packet_rssi(packet.packet_hash)
        reception_snr  = reticulum.get_packet_snr(packet.packet_hash)

        if reception_rssi != None:
            reception_stats += " [RSSI "+str(reception_rssi)+" dBm]"
        
        if reception_snr != None:
            reception_stats += " [SNR "+str(reception_snr)+" dBm]"

    else:
        if packet.rssi != None:
            reception_stats += " [RSSI "+str(packet.rssi)+" dBm]"
        
        if packet.snr != None:
            reception_stats += " [SNR "+str(packet.snr)+" dB]"

    RNS.log("Received packet from echo client, proof sent"+reception_stats)


##########################################################
#### Клиентская Часть ###################################
##########################################################

# Эта инициализация выполняется, когда пользователь
# выбирает запуск в режиме клиента
def client(destination_hexhash, configpath, timeout=None):
    global reticulum
    
    # Нам нужно двоичное представление хеша назначения,
    # который был введён в командной строке
    try:
        dest_len = (RNS.Reticulum.TRUNCATED_HASHLENGTH//8)*2
        if len(destination_hexhash) != dest_len:
            raise ValueError(
                "Destination length is invalid, must be {hex} hexadecimal characters ({byte} bytes).".format(hex=dest_len, byte=dest_len//2)
            )

        destination_hash = bytes.fromhex(destination_hexhash)
    except Exception as e:
        RNS.log("Invalid destination entered. Check your input!")
        RNS.log(str(e)+"\n")
        sys.exit(0)

    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)

    # Переопределяем уровень логирования для вывода
    # сообщений при получении объявления
    if RNS.loglevel < RNS.LOG_INFO:
        RNS.loglevel = RNS.LOG_INFO

    # Сообщаем пользователю, что клиент готов!
    RNS.log(
        "Echo client ready, hit enter to send echo request to "+
        destination_hexhash+
        " (Ctrl-C to quit)"
    )

    # Входим в цикл, работающий до выхода пользователя.
    # Если пользователь нажимает enter, мы попытаемся
    # отправить эхо-запрос на назначение, указанное в
    # командной строке.
    while True:
        input()
        
        # Сначала проверим, знает ли RNS путь к назначению.
        # Если знает, загрузим идентичность сервера и создадим
        # пакет
        if RNS.Transport.has_path(destination_hash):

            # Чтобы адресовать сервер, нам нужно знать его
            # открытый ключ, поэтому проверяем, знает ли
            # Reticulum это назначение. Это делается вызовом
            # метода "recall" модуля Identity. Если назначение
            # известно, он вернёт экземпляр Identity, который
            # можно использовать в исходящих назначениях.
            server_identity = RNS.Identity.recall(destination_hash)

            # Мы получили правильный экземпляр идентичности от
            # метода recall, так что создаём исходящее
            # назначение. Мы используем соглашение об именах:
            # example_utilities.echo.request, что соответствует
            # именованию, указанному в серверной части кода.
            request_destination = RNS.Destination(
                server_identity,
                RNS.Destination.OUT,
                RNS.Destination.SINGLE,
                APP_NAME,
                "echo",
                "request"
            )

            # Назначение готово, поэтому создаём пакет.
            # Устанавливаем назначение в только что созданный
            # request_destination, а в качестве данных добавляем
            # случайный хеш.
            echo_request = RNS.Packet(request_destination, RNS.Identity.get_random_hash())

            # Отправляем пакет! Если пакет успешно отправлен,
            # он вернёт экземпляр PacketReceipt.
            packet_receipt = echo_request.send()

            # Если пользователь указал таймаут, мы устанавливаем
            # этот таймаут на получение пакета и настраиваем
            # callback-функцию, которая будет вызвана, если
            # время ожидания истечёт.
            if timeout != None:
                packet_receipt.set_timeout(timeout)
                packet_receipt.set_timeout_callback(packet_timed_out)

            # Затем мы можем установить callback доставки на
            # получение. Он будет автоматически вызван, когда
            # доказательство для этого конкретного пакета будет
            # получено от назначения.
            packet_receipt.set_delivery_callback(packet_delivered)

            # Сообщаем пользователю, что эхо-запрос был отправлен
            RNS.log("Sent echo request to "+RNS.prettyhexrep(request_destination.hash))
        else:
            # Если мы не знаем это назначение, просим
            # пользователя подождать поступления объявления.
            RNS.log("Destination is not yet known. Requesting path...")
            RNS.log("Hit enter to manually retry once an announce is received.")
            RNS.Transport.request_path(destination_hash)

# Эта функция вызывается, когда наше ответное назначение
# получает пакет с доказательством.
def packet_delivered(receipt):
    global reticulum

    if receipt.status == RNS.PacketReceipt.DELIVERED:
        rtt = receipt.get_rtt()
        if (rtt >= 1):
            rtt = round(rtt, 3)
            rttstring = str(rtt)+" seconds"
        else:
            rtt = round(rtt*1000, 3)
            rttstring = str(rtt)+" milliseconds"

        reception_stats = ""
        if reticulum.is_connected_to_shared_instance:
            reception_rssi = reticulum.get_packet_rssi(receipt.proof_packet.packet_hash)
            reception_snr  = reticulum.get_packet_snr(receipt.proof_packet.packet_hash)

            if reception_rssi != None:
                reception_stats += " [RSSI "+str(reception_rssi)+" dBm]"
            
            if reception_snr != None:
                reception_stats += " [SNR "+str(reception_snr)+" dB]"

        else:
            if receipt.proof_packet != None:
                if receipt.proof_packet.rssi != None:
                    reception_stats += " [RSSI "+str(receipt.proof_packet.rssi)+" dBm]"
                
                if receipt.proof_packet.snr != None:
                    reception_stats += " [SNR "+str(receipt.proof_packet.snr)+" dB]"

        RNS.log(
            "Valid reply received from "+
            RNS.prettyhexrep(receipt.destination.hash)+
            ", round-trip time is "+rttstring+
            reception_stats
        )

# Эта функция вызывается, если время ожидания пакета
# истекло.
def packet_timed_out(receipt):
    if receipt.status == RNS.PacketReceipt.FAILED:
        RNS.log("Packet "+RNS.prettyhexrep(receipt.hash)+" timed out")

    # Получаем двоичное представление хеша из аргумента
    try:
        dest_len = (RNS.Reticulum.TRUNCATED_HASHLENGTH//8)*2
        if len(destination_hexhash) != dest_len:
            raise ValueError(
                "Destination length is invalid, must be {hex} hexadecimal characters ({byte} bytes).".format(hex=dest_len, byte=dest_len//2)
            )

        destination_hash = bytes.fromhex(destination_hexhash)
    except:
        RNS.log("Invalid destination entered. Check your input!\n")
        sys.exit(0)

    # Инициализируем Reticulum
    reticulum = RNS.Reticulum(configpath)

    # Проверяем, знаем ли мы путь к назначению
    if not RNS.Transport.has_path(destination_hash):
        RNS.log("Destination is not yet known. Requesting path and waiting for announce to arrive...")
        RNS.Transport.request_path(destination_hash)
        while not RNS.Transport.has_path(destination_hash):
            time.sleep(0.1)

    server_identity = RNS.Identity.recall(destination_hash)

    # Сообщаем о начале подключения
    RNS.log("Establishing link with server...")

        server_identity,
        RNS.Destination.OUT,
        RNS.Destination.SINGLE,
        APP_NAME,
        "linkexample"
    )

    link.set_link_established_callback(link_established)

    # Запускаем пользовательский цикл
    client_loop()

            text = input()

                        RNS.LOG_ERROR
                    )

            RNS.log("Error while sending data over the link: "+str(e))
            should_quit = True
            server_link.teardown()

# назначением и передавать структурированные сообщения  #
# по ней, используя канал.                              #

import os

##########################################################

# Данные канала должны быть структурированы в подклассе
# MessageBase. Это гарантирует, что канал сможет
# сериализовать и десериализовать объект, а также
# мультиплексировать его с другими объектами. Обеим
# сторонам связи нужны одинаковые определения объектов
# для возможности общения по каналу.
#
# Примечание: Объекты, которые мы хотим использовать
# через канал, должны быть зарегистрированы в канале,
# и у каждой связи свой экземпляр канала. Смотрите
# функции client_connected и link_established в этом
# примере, чтобы понять, как регистрируются типы
# сообщений.

# Создадим простой класс сообщений StringMessage,
# который будет передавать строку с временной меткой.

class StringMessage(RNS.MessageBase):
    # Переменной класса MSGTYPE должно быть присвоено
    # 2-байтовое целочисленное значение. Этот
    # идентификатор позволяет каналу найти конструктор
    # вашего сообщения при его поступлении по каналу.
    #
    # MSGTYPE должен быть уникальным среди всех типов
    # сообщений, которые мы регистрируем в канале.
    # MSGTYPEs >= 0xf000 зарезервированы для системы.
    MSGTYPE = 0x0101

    # Конструктор нашего объекта должен быть вызываемым
    # без аргументов. Мы можем иметь параметры, но они
    # должны иметь значение по умолчанию.
    #
    # Это необходимо, чтобы канал мог создать пустую
    # версию нашего сообщения, в которую можно будет
    # распаковать входящее сообщение.
    def __init__(self, data=None):
        self.data = data
        self.timestamp = datetime.now()
    # Наконец, нашему сообщению нужно реализовать
    # функции, которые канал сможет вызывать для упаковки
    # и распаковки нашего сообщения в/из сырой полезной
    # нагрузки пакета. Мы будем использовать пакет
    # umsgpack, входящий в состав RNS. Также можно
    # использовать пакет struct из Python, если нужен
    # больший контроль над структурой упакованных байт.
    #
    # Также обратите внимание, что упакованные объекты
    # сообщений должны полностью помещаться в один пакет.
    # Количество байт, доступных для полезной нагрузки
    # сообщений, можно запросить у канала, используя
    # свойство Channel.MDU. MDU канала немного меньше
    # MDU связи из-за кодирования заголовка сообщения.

    # Функция pack кодирует содержимое сообщения в
    # поток байтов.
    def pack(self) -> bytes:
        return umsgpack.packb((self.data, self.timestamp))

    # А функция unpack декодирует поток байтов обратно
    # в содержимое сообщения.
    def unpack(self, raw):
        self.data, self.timestamp = umsgpack.unpackb(raw)

Буфер (Buffer)¶

Пример Buffer показывает, как использовать буферизованные читатели и писатели для отправки бинарных данных между участниками Link.

##########################################################
# Этот пример RNS показывает, как настроить связь с     #

##########################################################

# поэтому используется пространство "example_utilities"
APP_NAME = "example_utilities"

#### Серверная Часть ####################################
##########################################################

# Ссылка на последний объект буфера

# выбирает запуск в качестве сервера
def server(configpath):
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)

    # Случайным образом создаём новую идентичность для
    # нашего примера
    server_identity = RNS.Identity()

    # подключаться. Мы хотим, чтобы клиенты создавали
    # связи с этим назначением, поэтому нам нужен тип
    # "single".
    server_destination = RNS.Destination(

    # Ждём запросов клиентов или ввода пользователя
    server_loop(server_destination)

    # Сообщаем пользователю, что всё готово
    RNS.log(
        "Link buffer example "+
        RNS.prettyhexrep(destination.hash)+
        " running, waiting for a connection."
    )

    RNS.log("Hit enter to manually send an announce (Ctrl-C to quit)")

    # Входим в цикл, работающий до выхода пользователя.
    # Если пользователь нажимает enter, мы объявим наше
    # серверное назначение в сети, что позволит клиентам
    # узнать, как создавать сообщения, направленные к нему.
    while True:
        entered = input()
        destination.announce()
        RNS.log("Sent announce from "+RNS.prettyhexrep(destination.hash))

# назначением, эта функция будет вызвана со ссылкой
# на связь.
def client_connected(link):
    global latest_client_link, latest_buffer
    latest_client_link = link

    RNS.log("Client connected")
    link.set_link_closed_callback(client_disconnected)

    # Если получено новое соединение, старый читатель
    # должен быть отключён.
    if latest_buffer:
        latest_buffer.close()


    # Создаём буферные объекты.
    #   Параметр stream_id в этих функциях - это что-то
    #   вроде файлового дескриптора, за исключением того,
    #   что он уникален для *получателя*.
    #
    #   В этом примере и читатель, и писатель используют
    #   stream_id = 0, но на самом деле существуют два
    #   отдельных однонаправленных потока, направленных в
    #   противоположные стороны.
    #
    channel = link.get_channel()
    latest_buffer = RNS.Buffer.create_bidirectional_buffer(0, 0, channel, server_buffer_ready)

def client_disconnected(link):
    RNS.log("Client disconnected")

def server_buffer_ready(ready_bytes: int):
    """
    Callback из буфера, когда буфер содержит данные

    :param ready_bytes: Количество байтов, готовых к чтению
    """
    global latest_buffer

    data = latest_buffer.read(ready_bytes)
    data = data.decode("utf-8")

    RNS.log("Received data over the buffer: " + data)

    reply_message = "I received \""+data+"\" over the buffer"
    reply_message = reply_message.encode("utf-8")
    latest_buffer.write(reply_message)
    latest_buffer.flush()

def client(destination_hexhash, configpath):
    # Нам нужно двоичное представление хеша назначения,
    # который был введён в командной строке
    try:
        dest_len = (RNS.Reticulum.TRUNCATED_HASHLENGTH//8)*2
        if len(destination_hexhash) != dest_len:
            raise ValueError(
                "Destination length is invalid, must be {hex} hexadecimal characters ({byte} bytes).".format(hex=dest_len, byte=dest_len//2)
            )

        destination_hash = bytes.fromhex(destination_hexhash)
    except:
        RNS.log("Invalid destination entered. Check your input!\n")
        sys.exit(0)
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)
    # Проверяем, известен ли нам путь к назначению
    if not RNS.Transport.has_path(destination_hash):
      RNS.log("Назначение ещё не известно. Запрашиваю путь и ожидаю прибытия анонса...")
      RNS.Transport.request_path(destination_hash)
      while not RNS.Transport.has_path(destination_hash):
          time.sleep(0.1)

    # Восстанавливаем идентичность сервера
    server_identity = RNS.Identity.recall(destination_hash)

    # Информируем пользователя, что начинаем подключение
    RNS.log("Устанавливаю связь с сервером...")

    # Когда идентичность сервера известна, мы настраиваем
    # назначение

        server_identity,
        RNS.Destination.OUT,
        RNS.Destination.SINGLE,
        APP_NAME,
        "bufferexample"
    )

    # И создаём линк
    link = RNS.Link(server_destination)

    # Также настроим функции для информирования
    # пользователя, когда линк установлен или закрыт
    link.set_link_established_callback(link_established)
    link.set_link_closed_callback(link_closed)

    # Всё настроено, переходим в цикл
    # для взаимодействия пользователя с примером
    client_loop()

def client_loop():
    global server_link

    # Ждём, пока линк не станет активным
    while not server_link:
        time.sleep(0.1)

    should_quit = False
    while not should_quit:
        try:
            print("> ", end=" ")
            text = input()

            # Проверяем, нужно ли выйти из примера
            if text == "quit" or text == "q" or text == "exit":
                should_quit = True
                server_link.teardown()
            else:
                # Иначе кодируем текст и записываем его в буфер.
                text = text.encode("utf-8")
                buffer.write(text)
                # Сбрасываем буфер, чтобы принудительно отправить данные.
                buffer.flush()


        except Exception as e:
            RNS.log("Ошибка при отправке данных через буфер линка: "+str(e))
            should_quit = True
            server_link.teardown()

Передача файлов¶

Пример Передача файлов реализует базовую программу файлового сервера, которая позволяет клиентам подключаться и скачивать файлы. Программа использует интерфейс Resource для эффективной передачи файлов любого размера через Линк Reticulum.

##########################################################
# Этот пример RNS демонстрирует простую программу       #
# файлового сервера и клиента. Сервер будет раздавать   #
# директорию с файлами, а клиенты могут просматривать   #
# и скачивать файлы с сервера.                          #
#                                                       #
# Обратите внимание, что использование RNS Resources    #
# для передачи больших файлов не рекомендуется,         #
# поскольку сжатие, шифрование и последовательное       #
# хеширование могут занять много времени на системах    #
# с медленными ЦП, что, вероятно, приведёт к тайм-ауту  #
# клиента до того, как отправитель ресурса              #
# завершит подготовку ресурса.                          #
#                                                       #
# Если вам нужно передавать большие файлы, используйте  #
# класс Bundle, который автоматически нарезает данные   #
# на части, подходящие для упаковки в Resource.         #
##########################################################

import os
import sys
import time
import threading
import argparse
import RNS
import RNS.vendor.umsgpack as umsgpack

# Давайте определим имя приложения. Мы будем использовать его для всех
# создаваемых назначений. Поскольку этот пример эхо
# является частью набора примеров утилит, мы поместим
# их все в пространство имён приложения "example_utilities"
APP_NAME = "example_utilities"

# Также определим тайм-аут по умолчанию в секундах
APP_TIMEOUT = 45.0

##########################################################
#### Серверная часть ####################################
##########################################################

serve_path = None

# Эта инициализация выполняется, когда пользователь выбирает
# запуск в качестве сервера
def server(configpath, path):
    # Сначала мы должны инициализировать Reticulum
    reticulum = RNS.Reticulum(configpath)

    # Случайным образом создаём новую идентичность для нашего файлового сервера
    server_identity = RNS.Identity()

    global serve_path
    serve_path = path

    # Мы создаём назначение, к которому клиенты могут подключаться. Мы
    # хотим, чтобы клиенты создавали линки к этому назначению, поэтому нам
    # нужно создать тип назначения "single".
    server_destination = RNS.Destination(
        server_identity,
        RNS.Destination.IN,
        RNS.Destination.SINGLE,
        APP_NAME,
        "filetransfer",
        "server"
    )

    # Мы настраиваем функцию, которая будет вызываться каждый раз,
    # когда новый клиент создаёт линк к этому назначению.
    server_destination.set_link_established_callback(client_connected)

    # Всё готово!
    # Будем ждать клиентских запросов или пользовательского ввода
    announceLoop(server_destination)

def announceLoop(destination):
    # Сообщим пользователю, что всё готово
    RNS.log("Файловый сервер "+RNS.prettyhexrep(destination.hash)+" запущен")
    RNS.log("Нажмите Enter для ручной отправки анонса (Ctrl-C для выхода)")

    # Мы входим в цикл, который выполняется, пока пользователь не выйдет.
    # Если пользователь нажмёт Enter, мы анонсируем назначение
    # нашего сервера в сеть, что позволит клиентам
    # узнать, как создавать сообщения, направленные к нему.
    while True:
        entered = input()
        destination.announce()
        RNS.log("Отправлен анонс от "+RNS.prettyhexrep(destination.hash))

# Вот удобная функция для вывода списка всех файлов
# в нашей раздаваемой директории
def list_files():
    # Мы добавляем все записи из директории, которые
    # являются файлами и не начинаются с "."
    global serve_path
    return [file for file in os.listdir(serve_path) if os.path.isfile(os.path.join(serve_path, file)) and file[:1] != "."]

# Когда клиент устанавливает линк с назначением
# нашего сервера, вызывается эта функция со
# ссылкой на линк. Затем мы отправляем клиенту
# список файлов, размещённых на сервере.
def client_connected(link):
    # Проверяем, существует ли ещё раздаваемая директория
    if os.path.isdir(serve_path):
        RNS.log("Клиент подключен, отправляю список файлов...")

        link.set_link_closed_callback(client_disconnected)

        # Мы упаковываем список файлов для отправки в пакете
        data = umsgpack.packb(list_files())

        # Проверяем размер упакованных данных
        if len(data) <= RNS.Link.MDU:
            # Если данные помещаются в один пакет, мы просто
            # отправим их как один пакет по линку.
            list_packet = RNS.Packet(link, data)
            list_receipt = list_packet.send()
            list_receipt.set_timeout(APP_TIMEOUT)
            list_receipt.set_delivery_callback(list_delivered)
            list_receipt.set_timeout_callback(list_timeout)
        else:
            RNS.log("Слишком много файлов в раздаваемой директории!", RNS.LOG_ERROR)
            RNS.log("Вам следует реализовать функцию для разделения списка файлов на несколько пакетов.", RNS.LOG_ERROR)
            RNS.log("Подсказка: Клиент уже поддерживает это :)", RNS.LOG_ERROR)

        # После этого мы просто будем держать линк открытым
        # до тех пор, пока клиент не запросит файл. Мы
        # настроим функцию, которая вызывается, когда
        # клиент отправляет пакет с запросом файла.
        link.set_packet_callback(client_request)
    else:
        RNS.log("Клиент подключился, но раздаваемый путь больше не существует!", RNS.LOG_ERROR)
        link.teardown()

def client_disconnected(link):
    RNS.log("Клиент отключился")

def client_request(message, packet):
    global serve_path

    try:
        filename = message.decode("utf-8")
    except Exception as e:
        filename = None

    if filename in list_files():
        try:
            # Если у нас есть запрошенный файл, мы
            # прочитаем его и упакуем как ресурс
            RNS.log("Клиент запросил \""+filename+"\"")
            file = open(os.path.join(serve_path, filename), "rb")

            file_resource = RNS.Resource(
                file,
                packet.link,
                callback=resource_sending_concluded
            )

            file_resource.filename = filename
        except Exception as e:
            # Если что-то пошло не так, мы закрываем
            # линк
            RNS.log("Ошибка при чтении файла \""+filename+"\"", RNS.LOG_ERROR)
            packet.link.teardown()
            raise e
    else:
        # Если у нас его нет, мы закрываем линк
        RNS.log("Клиент запросил неизвестный файл")
        packet.link.teardown()

# Эта функция вызывается на сервере, когда
# передача ресурса завершается.
def resource_sending_concluded(resource):
    if hasattr(resource, "filename"):
        name = resource.filename
    else:
        name = "ресурс"

    if resource.status == RNS.Resource.COMPLETE:
        RNS.log("Отправка \""+name+"\" клиенту завершена")
    elif resource.status == RNS.Resource.FAILED:
        RNS.log("Отправка \""+name+"\" клиенту не удалась")

def list_delivered(receipt):
    RNS.log("Список файлов был получен клиентом")

def list_timeout(receipt):
    RNS.log("Время отправки списка клиенту истекло, закрываю этот линк")
    link = receipt.destination
    link.teardown()

server_files      = []

# Ссылка на линк сервера
server_link       = None

# И ссылка на текущую загрузку
current_download  = None
current_filename  = None

download_started  = 0

    # И создаём линк
    link = RNS.Link(server_destination)

    # Мы ожидаем, что любые обычные пакеты данных в линке
    # будут содержать список раздаваемых файлов, поэтому мы устанавливаем
    # соответствующий обратный вызов
    link.set_packet_callback(filelist_received)

    # Также настроим функции для информирования
    # пользователя, когда линк установлен или закрыт
    link.set_link_established_callback(link_established)
    link.set_link_closed_callback(link_closed)

    # И настраиваем линк на автоматическое начало
    # загрузки анонсированных ресурсов
    link.set_resource_strategy(RNS.Link.ACCEPT_ALL)
    link.set_resource_started_callback(download_began)
    link.set_resource_concluded_callback(download_concluded)

    menu()

# Запрашивает указанный файл с сервера
def download(filename):
    global server_link, menu_mode, current_filename, transfer_size, download_started
    current_filename = filename
    download_started = 0
    transfer_size    = 0

    # Мы просто создаём пакет, содержащий
    # запрашиваемое имя файла, и отправляем его вниз
    # по линку. Мы также указываем, что нам не нужна
    # квитанция о получении пакета.
    request_packet = RNS.Packet(server_link, filename.encode("utf-8"), create_receipt=False)
    request_packet.send()

    print("")
    print(("Запрошен \""+filename+"\" с сервера, ожидаю начала загрузки..."))
    menu_mode = "download_started"

def size_str(num, suffix='B'):
    units = ['','Ki','Mi','Gi','Ti','Pi','Ei','Zi']
    last_unit = 'Yi'

    if suffix == 'b':
        num *= 8
        units = ['','K','M','G','T','P','E','Z']
        last_unit = 'Y'

    for unit in units:
        if abs(num) < 1024.0:
            return "%3.2f %s%s" % (num, unit, suffix)
        num /= 1024.0
    return "%.2f %s%s" % (num, last_unit, suffix)

Пользовательские интерфейсы¶

Пример ExampleInterface демонстрирует создание пользовательских интерфейсов для Reticulum. Любое количество пользовательских интерфейсов может быть загружено и использовано Reticulum, и они будут полностью наравне с нативно включенными интерфейсами, включая все поддерживаемые режимы интерфейса и общие параметры конфигурации.

# Этот пример иллюстрирует создание пользовательского
# определения интерфейса, которое может быть загружено
# и использовано Reticulum во время выполнения. Может быть
# создано и загружено любое количество пользовательских
# интерфейсов. Чтобы использовать интерфейс, поместите его
# в папку ~/.reticulum/interfaces и добавьте запись
# интерфейса в ваш файл конфигурации Reticulum, подобную этой:

#  [[Example Custom Interface]]
#    type = ExampleInterface
#    enabled = no
#    mode = gateway
#    port = /dev/ttyUSB0
#    speed = 115200
#    databits = 8
#    parity = none
#    stopbits = 1

from time import sleep
import sys
import threading
import time

# Этот вспомогательный класс HDLC используется интерфейсом
# для разделения и пакетирования данных через физическую
# среду - в данном случае последовательное соединение.
class HDLC():
    # Этот пример интерфейса пакетирует данные, используя
    # упрощенное кадрирование HDLC, подобное PPP
    FLAG     = 0x7E
    ESC      = 0x7D
    ESC_MASK = 0x20

    @staticmethod
    def escape(data):
        data = data.replace(bytes([HDLC.ESC]), bytes([HDLC.ESC, HDLC.ESC^HDLC.ESC_MASK]))
        data = data.replace(bytes([HDLC.FLAG]), bytes([HDLC.ESC, HDLC.FLAG^HDLC.ESC_MASK]))
        return data

# Давайте определим наш класс пользовательского интерфейса.
# Он должен быть подклассом класса RNS "Interface".
class ExampleInterface(Interface):
    # Все классы интерфейсов должны определять размер
    # IFAC по умолчанию, используемый при настройке IFAC,
    # когда пользователь не указал свой размер IFAC. Этот
    # параметр указывается в байтах.
    DEFAULT_IFAC_SIZE = 8

    # Следующие свойства являются локальными для этой
    # конкретной реализации интерфейса.
    owner    = None
    port     = None
    speed    = None
    databits = None
    parity   = None
    stopbits = None
    serial   = None

    # Все интерфейсы Reticulum должны иметь метод __init__,
    # который принимает 2 позиционных аргумента:
    # Экземпляр-владелец RNS Transport и словарь
    # со значениями конфигурации.
    def __init__(self, owner, configuration):

        # Следующие строки демонстрируют обработку
        # потенциальных зависимостей, необходимых для
        # корректной работы интерфейса.
        import importlib
        if importlib.util.find_spec('serial') != None:
            import serial
        else:
            RNS.log("Для использования этого интерфейса необходимо установить модуль последовательной связи.", RNS.LOG_CRITICAL)
            RNS.log("Вы можете установить его командой: python3 -m pip install pyserial", RNS.LOG_CRITICAL)
            RNS.panic()

        # Начинаем с инициализации суперкласса
        super().__init__()

        # Чтобы убедиться, что данные конфигурации в правильном
        # формате, мы разбираем их с помощью следующего метода
        # общего класса Interface. Этот шаг необходим для
        # обеспечения совместимости на всех платформах,
        # которые поддерживает Reticulum.
        ifconf    = Interface.get_config_obj(configuration)

        # Читаем имя интерфейса из конфигурации
        # и устанавливаем его в нашем экземпляре интерфейса.
        name      = ifconf["name"]
        self.name = name

        # Мы читаем параметры конфигурации из предоставленных
        # данных конфигурации и предоставляем значения по умолчанию
        # на случай, если какие-либо отсутствуют.
        port      = ifconf["port"] if "port" in ifconf else None
        speed     = int(ifconf["speed"]) if "speed" in ifconf else 9600
        databits  = int(ifconf["databits"]) if "databits" in ifconf else 8
        parity    = ifconf["parity"] if "parity" in ifconf else "N"
        stopbits  = int(ifconf["stopbits"]) if "stopbits" in ifconf else 1

        # В случае, если порт не указан, мы прерываем настройку,
        # вызывая исключение.
        if port == None:
            raise ValueError(f"Не указан порт для {self}")

        # Все интерфейсы должны предоставлять значение аппаратного
        # MTU экземпляру RNS Transport. Это значение должно быть
        # максимальным размером полезной нагрузки пакета данных,
        # который базовая среда способна обработать во всех
        # случаях без какой-либо сегментации.
        self.HW_MTU = 564

        # Мы изначально устанавливаем свойство "online" в false,
        # поскольку интерфейс еще не был полностью
        # инициализирован и подключен.
        self.online   = False

        # В этом случае мы также можем установить указанную
        # скорость передачи данных интерфейса на скорость
        # последовательного порта.
        self.bitrate  = speed

        # Настраиваем внутренние свойства интерфейса
        # в соответствии с предоставленной конфигурацией.
        self.pyserial = serial
        self.serial   = None
        self.owner    = owner
        self.port     = port
        self.speed    = speed
        self.databits = databits
        self.parity   = serial.PARITY_NONE
        self.stopbits = stopbits
        self.timeout  = 100

        if parity.lower() == "e" or parity.lower() == "even":
            self.parity = serial.PARITY_EVEN

        if parity.lower() == "o" or parity.lower() == "odd":
            self.parity = serial.PARITY_ODD

        # Поскольку все необходимые параметры теперь настроены,
        # мы попытаемся открыть последовательный порт.
        try:
            self.open_port()
        except Exception as e:
            RNS.log("Не удалось открыть последовательный порт для интерфейса "+str(self), RNS.LOG_ERROR)
            raise e

        # Если открытие порта прошло успешно, выполняем любую
        # необходимую пост-открытие конфигурацию.
        if self.serial.is_open:
            self.configure_device()
        else:
            raise IOError("Не удалось открыть последовательный порт")

    # Открываем последовательный порт с предоставленными
    # параметрами конфигурации и сохраняем ссылку на открытый порт.
    def open_port(self):
        RNS.log("Открываю последовательный порт "+self.port+"...", RNS.LOG_VERBOSE)
        self.serial = self.pyserial.Serial(
            port = self.port,
            baudrate = self.speed,
            bytesize = self.databits,
            parity = self.parity,
            stopbits = self.stopbits,
            xonxoff = False,
            rtscts = False,
            timeout = 0,
            inter_byte_timeout = None,
            write_timeout = None,
            dsrdtr = False,
        )

    # Единственное, что требуется после открытия порта, —
    # это подождать небольшое количество времени для
    # инициализации оборудования, а затем запустить поток,
    # который читает любые входящие данные с устройства.
    def configure_device(self):
        sleep(0.5)
        thread = threading.Thread(target=self.read_loop)
        thread.daemon = True
        thread.start()
        self.online = True
        RNS.log("Последовательный порт "+self.port+" теперь открыт", RNS.LOG_VERBOSE)


    # Этот метод будет вызываться из нашего цикла чтения
    # всякий раз, когда полный пакет был получен через
    # базовую среду.
    def process_incoming(self, data):
        # Обновляем наш счетчик полученных байтов
        self.rxb += len(data)

        # И отправляем пакет данных экземпляру Transport
        # для обработки.
        self.owner.inbound(data, self)

    # Запущенный экземпляр Reticulum Transport будет
    # вызывать этот метод на интерфейсе всякий раз, когда
    # интерфейс должен передать пакет.
    def process_outgoing(self,data):
        if self.online:
            # Сначала экранируем и пакетируем данные
            # в соответствии с кадрированием HDLC.
            data = bytes([HDLC.FLAG])+HDLC.escape(data)+bytes([HDLC.FLAG])

            # Затем записываем кадрированные данные в порт
            written = self.serial.write(data)

            # Обновляем счетчик переданных байтов
            # и убеждаемся, что все данные были записаны
            self.txb += len(data)
            if written != len(data):
                raise IOError("Последовательный интерфейс записал только "+str(written)+" байт из "+str(len(data)))

    # Этот цикл чтения выполняется в потоке и непрерывно
    # принимает байты из нижележащего последовательного порта.
    # Когда полный пакет получен, он будет
    # отправлен в метод process_incoming, который
    # в свою очередь передаст его экземпляру Transport.
    def read_loop(self):
        try:
            in_frame = False
            escape = False
            data_buffer = b""
            last_read_ms = int(time.time()*1000)

            while self.serial.is_open:
                if self.serial.in_waiting:
                    byte = ord(self.serial.read(1))
                    last_read_ms = int(time.time()*1000)

                    if (in_frame and byte == HDLC.FLAG):
                        in_frame = False
                        self.process_incoming(data_buffer)
                    elif (byte == HDLC.FLAG):
                        in_frame = True
                        data_buffer = b""
                    elif (in_frame and len(data_buffer) < self.HW_MTU):
                        if (byte == HDLC.ESC):
                            escape = True
                        else:
                            if (escape):
                                if (byte == HDLC.FLAG ^ HDLC.ESC_MASK):
                                    byte = HDLC.FLAG
                                if (byte == HDLC.ESC  ^ HDLC.ESC_MASK):
                                    byte = HDLC.ESC
                                escape = False
                            data_buffer = data_buffer+bytes([byte])

                else:
                    time_since_last = int(time.time()*1000) - last_read_ms
                    if len(data_buffer) > 0 and time_since_last > self.timeout:
                        data_buffer = b""
                        in_frame = False
                        escape = False
                    sleep(0.08)

        except Exception as e:
            self.online = False
            RNS.log("Произошла ошибка последовательного порта, содержащееся исключение: "+str(e), RNS.LOG_ERROR)
            RNS.log("Интерфейс "+str(self)+" столкнулся с невосстановимой ошибкой и теперь не в сети.", RNS.LOG_ERROR)

            if RNS.Reticulum.panic_on_interface_error:
                RNS.panic()

            RNS.log("Reticulum будет периодически пытаться переподключить интерфейс.", RNS.LOG_ERROR)

        self.online = False
        self.serial.close()
        self.reconnect_port()

    # Этот метод обрабатывает отключения последовательного порта.
    def reconnect_port(self):
        while not self.online:
            try:
                time.sleep(5)
                RNS.log("Попытка переподключения последовательного порта "+str(self.port)+" для "+str(self)+"...", RNS.LOG_VERBOSE)
                self.open_port()
                if self.serial.is_open:
                    self.configure_device()
            except Exception as e:
                RNS.log("Ошибка при переподключении порта, содержащееся исключение: "+str(e), RNS.LOG_ERROR)

        RNS.log("Переподключен последовательный порт для "+str(self))

    # Сигнализируем Reticulum, что этот интерфейс не должен
    # выполнять какое-либо ограничение входящего трафика.
    def should_ingress_limit(self):
        return False

    # Мы должны предоставить строковое представление этого
    # интерфейса, которое используется всякий раз, когда интерфейс
    # выводится в журналах или внешних программах.
    def __str__(self):
        return "ExampleInterface["+self.name+"]"

# Наконец, регистрируем определенный класс интерфейса как
# целевой класс для использования Reticulum в качестве интерфейса
interface_class = ExampleInterface
